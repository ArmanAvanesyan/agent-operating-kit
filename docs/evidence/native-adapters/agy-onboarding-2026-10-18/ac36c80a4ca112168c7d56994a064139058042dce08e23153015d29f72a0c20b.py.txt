def _load_module(name, path):
    spec = importlib.util.spec_from_file_location(name, path)
    module = importlib.util.module_from_spec(spec)
    spec.loader.exec_module(module)
    return module

def sha(data):
    return hashlib.sha256(data).hexdigest()


def file_sha(path):
    h = hashlib.sha256()
    with Path(path).open("rb") as stream:
        for block in iter(lambda: stream.read(1024 * 1024), b""):
            h.update(block)
    return h.hexdigest()


def _json_path(value):
    return json.dumps(str(value))


def sandbox_profile(root, owned_cli, tty_device):
    """Use private file confinement, allow OAuth networking, deny OS keychain."""
    root = Path(root).resolve(strict=True)
    owned_cli = Path(owned_cli).resolve(strict=True)
    real_home = Path(pwd.getpwuid(os.getuid()).pw_dir).resolve()
    tty_device = str(tty_device)
    if not tty_device.startswith("/dev/tty"):
        raise ValueError("the allowed terminal device must be an allocated /dev/tty device")
    # Cursor's native file boundary shape with networking enabled for the
    # operator-approved OAuth flow. The copied vendor executable is excluded
    # from the otherwise-owned-root write exception. No Mach exception exists.
    return (
        "(version 1)(allow default)"
        f"(deny file-read* (subpath {_json_path(real_home)}))"
        "(deny file-write*)"
        f"(allow file-write* (subpath {_json_path(root)})"
        ' (literal "/dev/null") (literal "/dev/tty")'
        f" (literal {_json_path(tty_device)}))"
        f"(deny file-write* (subpath {_json_path(owned_cli)}))"
        '(deny mach-lookup (global-name "com.apple.securityd"))'
    )


def env_for(root):
    home = Path(root) / "home"
    tmp = Path(root) / "tmp"
    config = home / ".config"
    cache = home / ".cache"
    for directory in (home, tmp, config, cache):
        directory.mkdir(parents=True, exist_ok=True)
        directory.chmod(0o700)
    env = {
        "PATH": "/usr/bin:/bin",
        "HOME": str(home),
        "TMPDIR": str(tmp),
        "XDG_CONFIG_HOME": str(config),
        "XDG_CACHE_HOME": str(cache),
        "AGY_CLI_DISABLE_AUTO_UPDATE": "true",
        "BROWSER": "/usr/bin/open",
        "TERM": "xterm-256color",
        "PYTHONDONTWRITEBYTECODE": "1",
    }
    if MANUAL_OAUTH_MODE:
        env.update({"SSH_CONNECTION": "127.0.0.1 1 127.0.0.1 22", "SSH_CLIENT": "127.0.0.1 1 22", "SSH_TTY": "/dev/tty"})
    return env


def append_audit(path, event, **fields):
    record = {"event": event, "at": datetime.now(timezone.utc).isoformat(), **fields}
    with Path(path).open("a", encoding="utf-8") as stream:
        stream.write(json.dumps(record, sort_keys=True) + "\n")
        stream.flush()
        os.fsync(stream.fileno())


def _owned_dir(prefix):
    path = Path(tempfile.mkdtemp(prefix=prefix, dir=PRIVATE_TEMP_ROOT)).resolve(strict=True)
    path.chmod(0o700)
    return path


def _atomic_json(path, value):
    path = Path(path)
    temporary = path.with_name(path.name + ".tmp")
    with temporary.open("x", encoding="utf-8") as stream:
        json.dump(value, stream, sort_keys=True, indent=2)
        stream.write("\n")
        stream.flush()
        os.fsync(stream.fileno())
    temporary.chmod(0o600)
    os.replace(temporary, path)

def active_authorization_code_field(screen, text):
    """A title authorizes input only with instructions and the cursor in its box."""
    rows = text.splitlines()
    if not re.search(r"copy[^\n]*\bcode\b[^\n]*\bbrowser\b[^\n]*\bpaste\b[^\n]*\bbelow\b", text, re.I):
        return False
    for index, row in enumerate(rows):
        if not AUTHORIZATION_CODE_TITLE.fullmatch(row):
            continue
        if not index < screen.row <= index + 4 or screen.row >= len(rows):
            continue
        field = rows[screen.row]
        left = field.find("│")
        right = field.rfind("│")
        if 0 <= left < screen.col < right:
            return True
    return False

def _disable_terminal_echo(fd):
    if not os.isatty(fd):
        raise RuntimeError("authorization-code input requires a TTY")
    original = termios.tcgetattr(fd)
    disabled = list(original)
    disabled[3] &= ~(termios.ECHO | getattr(termios, "ECHONL", 0))
    termios.tcsetattr(fd, termios.TCSANOW, disabled)
    return original


def _restore_terminal(fd, original):
    if original is not None:
        termios.tcsetattr(fd, termios.TCSANOW, original)


def validate_authorization_code_file(path):
    """Validate a task-owned, private temporary file without reading its bytes."""
    path = Path(path).absolute()
    repo = REPO.resolve(strict=True)
    temp_root = Path(PRIVATE_TEMP_ROOT).resolve(strict=True)
    parent = path.parent
    try:
        parent_info = parent.lstat()
        file_info = path.lstat()
    except OSError:
        raise RuntimeError("authorization-code file is unavailable") from None
    if (parent.is_symlink() or parent.resolve(strict=True) != parent or
            not parent.is_relative_to(temp_root) or parent == temp_root or
            parent.is_relative_to(repo)):
        raise RuntimeError("authorization-code parent is outside the private temporary area")
    if (not stat.S_ISDIR(parent_info.st_mode) or parent_info.st_uid != os.getuid() or
            parent_info.st_mode & 0o077):
        raise RuntimeError("authorization-code parent permissions are unsafe")
    if (path.is_symlink() or not stat.S_ISREG(file_info.st_mode) or
            file_info.st_uid != os.getuid() or file_info.st_nlink != 1 or
            file_info.st_mode & 0o077 or file_info.st_size > MAX_AUTHORIZATION_CODE_BYTES + 1):
        raise RuntimeError("authorization-code file permissions or type are unsafe")
    return path


def read_authorization_code_file(path):
    """Read a complete one-line code using no-follow and inode checks."""
    path = validate_authorization_code_file(path)
    flags = os.O_RDONLY | getattr(os, "O_NOFOLLOW", 0)
    try:
        fd = os.open(path, flags)
    except OSError:
        raise RuntimeError("authorization-code file could not be opened safely") from None
    try:
        opened = os.fstat(fd)
        current = path.lstat()
        if (not stat.S_ISREG(opened.st_mode) or opened.st_dev != current.st_dev or
                opened.st_ino != current.st_ino or opened.st_uid != os.getuid() or
                opened.st_nlink != 1 or opened.st_mode & 0o077 or
                opened.st_size > MAX_AUTHORIZATION_CODE_BYTES + 1):
            raise RuntimeError("authorization-code file changed or exceeds limits")
        data = bytearray()
        while len(data) <= MAX_AUTHORIZATION_CODE_BYTES + 1:
            block = os.read(fd, MAX_AUTHORIZATION_CODE_BYTES + 2 - len(data))
            if not block:
                break
            data.extend(block)
        if len(data) > MAX_AUTHORIZATION_CODE_BYTES + 1:
            raise RuntimeError("authorization-code file exceeds limit")
        if not data:
            return None, (opened.st_dev, opened.st_ino)
        if data[-1] not in (10, 13):
            if any(value not in AUTHORIZATION_CODE_BYTES for value in data):
                raise RuntimeError("authorization-code file has invalid characters")
            return None, (opened.st_dev, opened.st_ino)
        if data.count(10) + data.count(13) != 1 or len(data) > MAX_AUTHORIZATION_CODE_BYTES + 1:
            raise RuntimeError("authorization-code file must contain one bounded line")
        if any(value not in AUTHORIZATION_CODE_BYTES for value in data[:-1]):
            raise RuntimeError("authorization-code file has invalid characters")
        return bytes(data), (opened.st_dev, opened.st_ino)
    finally:
        os.close(fd)


def remove_authorization_code_file(path):
    """Remove only the validated path entry; unlink never follows symlinks."""
    if path is None:
        return
    path = Path(path)
    try:
        repo = REPO.resolve(strict=True)
        temp_root = Path(PRIVATE_TEMP_ROOT).resolve(strict=True)
        parent = path.parent
        parent_info = parent.lstat()
        if (parent.is_symlink() or parent.resolve(strict=True) != parent or
                not parent.is_relative_to(temp_root) or parent == temp_root or
                parent.is_relative_to(repo) or not stat.S_ISDIR(parent_info.st_mode) or
                parent_info.st_uid != os.getuid() or parent_info.st_mode & 0o077):
            raise RuntimeError("authorization-code parent changed before cleanup")
        entry = path.lstat()
        # Unlinking a symlink removes only the link. Never open or follow it.
        if stat.S_ISLNK(entry.st_mode) or (stat.S_ISREG(entry.st_mode) and
                entry.st_uid == os.getuid() and entry.st_nlink == 1):
            path.unlink()
        else:
            raise RuntimeError("authorization-code file changed before cleanup")
    except FileNotFoundError:
        return
    except OSError:
        raise RuntimeError("authorization-code file cleanup failed") from None


def _unlink_code_before_submit(path, identity):
    try:
        path = validate_authorization_code_file(path)
        current = path.lstat()
        if (current.st_dev, current.st_ino) != identity:
            raise RuntimeError("authorization-code file changed before submission")
        path.unlink()
    except OSError:
        raise RuntimeError("authorization-code file could not be removed safely") from None


def _validate_and_submit_code(master, code_buffer, chunk, audit_path, authorization_code_fd=None,
                              *, before_submit=None):
    """Accept exactly one bounded ASCII code line; never report its value."""
    terminators = [index for index, value in enumerate(chunk) if value in (10, 13)]
    if len(terminators) > 1:
        raise RuntimeError("authorization-code input must be one line")
    if terminators:
        end = terminators[0]
        if end != len(chunk) - 1:
            raise RuntimeError("authorization-code input must be one line")
        code_buffer.extend(chunk[:end])
        complete = True
    else:
        code_buffer.extend(chunk)
        complete = False
    if len(code_buffer) > MAX_AUTHORIZATION_CODE_BYTES:
        raise RuntimeError("authorization-code input exceeds limit")
    if any(value not in AUTHORIZATION_CODE_BYTES for value in code_buffer):
        raise RuntimeError("authorization-code input has invalid characters")
    if complete:
        if not code_buffer:
            raise RuntimeError("authorization-code input is empty")
        if before_submit is not None:
            before_submit()
        os.write(master, bytes(code_buffer) + b"\r")
        append_audit(audit_path, "authorization-code-submitted", authorizationCodeSubmitted=True)
        if authorization_code_fd is not None:
            termios.tcflush(authorization_code_fd, termios.TCIFLUSH)
        code_buffer[:] = b"\x00" * len(code_buffer)
        return True
    return False

def open_manual_oauth_browser(wire):
    """Open only a complete native Google OAuth URL, without logging it."""
    for match in re.finditer(rb"https://accounts\.google\.com/[^\s\x1b\x07]{1,8192}(?=[\s\x1b\x07])", wire):
        try:
            url = match.group().decode("ascii")
            parsed = urlparse(url)
            query = parse_qs(parsed.query)
            if (parsed.scheme != "https" or parsed.netloc != "accounts.google.com" or
                    parsed.path not in ("/o/oauth2/auth", "/o/oauth2/v2/auth") or
                    not query.get("client_id") or query.get("response_type") != ["code"]):
                continue
            result = subprocess.run(["/usr/bin/open", url], stdout=subprocess.DEVNULL,
                                    stderr=subprocess.DEVNULL, timeout=10)
            return result.returncode == 0
        except (ValueError, UnicodeError, OSError, subprocess.TimeoutExpired):
            return False
    return None


def terminal_query_replies(wire, answered):
    """Answer exact terminal capability queries; never synthesize user input."""
    queries = (
        (b"\x1b[6n", "cursor-position", None),
        (b"\x1b[c", "primary-attributes", b"\x1b[?1;2c"),
        (b"\x1b[0c", "primary-attributes-explicit", b"\x1b[?1;2c"),
        (b"\x1b[>c", "secondary-attributes", b"\x1b[>0;276;0c"),
        (b"\x1b[>0c", "secondary-attributes-explicit", b"\x1b[>0;276;0c"),
        (b"\x1b[>q", "terminal-version", b"\x1bP>|AOK-probe\x1b\\"),
        (b"\x1b[?u", "keyboard-unsupported", b"\x1b[?0u"),
        (b"\x1b[?2026$p", "sync-unsupported", b"\x1b[?2026;0$y"),
        (b"\x1b[?2027$p", "unicode-width-unsupported", b"\x1b[?2027;0$y"),
        (b"\x1b[?1004$p", "focus-unsupported", b"\x1b[?1004;0$y"),
        (b"\x1b]10;?\x07", "foreground-color", b"\x1b]10;rgb:ffff/ffff/ffff\x07"),
        (b"\x1b]11;?\x07", "background-color", b"\x1b]11;rgb:0000/0000/0000\x07"),
    )
    replies = []
    for query, label, reply in queries:
        start = 0
        while True:
            index = wire.find(query, start)
            if index < 0:
                break
            start = index + len(query)
            identity = (index, label)
            if identity in answered:
                continue
            if reply is None:
                try:
                    screen = AGY._terminal.Screen(width=LOGIN_WIDTH, height=LOGIN_HEIGHT).feed(wire[:index])
                    screen.render()
                except AGY._terminal.TerminalAmbiguity:
                    continue
                response = ("\x1b[" + str(screen.row + 1) + ";" + str(screen.col + 1) + "R").encode("ascii")
            else:
                response = reply
            answered.add(identity)
            replies.append((label, response))
    return replies


def _capture_login(master, deadline, *, enter_sent, audit_path, case_root,
                   authorization_code_fd=None, authorization_code_file=None,
                   byte_limit=512 * 1024):
    """Capture in memory; optionally send one Enter at the explicit Google gate."""
    chunks = []
    answered_queries = set()
    browser_attempted = False
    total = 0
    last_state = None
    state = None
    code_buffer = bytearray()
    authorization_code_submitted = False
    color_scheme_continued = False
    last_change = time.monotonic()
    try:
        while time.monotonic() < deadline and total < byte_limit:
            can_read_code = (
                (authorization_code_fd is not None or authorization_code_file is not None) and
                not authorization_code_submitted and
                enter_sent and state is not None and
                state.get("authorizationCodePromptObserved") is True and
                state.get("terminalAmbiguityObserved") is False and
                state.get("localPromptObserved") is False
            )
            read_fds = [master]
            if can_read_code and authorization_code_fd is not None:
                read_fds.append(authorization_code_fd)
            ready, _, _ = select.select(
                read_fds, [], [], min(0.2, max(0.0, deadline - time.monotonic()))
            )
            if not ready:
                if state and state.get("terminalAmbiguityObserved") and time.monotonic() - last_change >= 2:
                    append_audit(audit_path, "stable-terminal-ambiguity", state=_safe_state(state))
                    return state, enter_sent
            if master in ready:
                try:
                    block = os.read(master, min(8192, byte_limit - total))
                except BlockingIOError:
                    block = None
                except OSError:
                    break
                if block == b"":
                    break
                if block:
                    chunks.append(block)
                    total += len(block)
                    wire = b"".join(chunks)
                    if MANUAL_OAUTH_MODE and enter_sent and not browser_attempted:
                        opened = open_manual_oauth_browser(wire)
                        if opened is not None:
                            browser_attempted = True
                            append_audit(audit_path, "manual-oauth-browser-launch", success=opened)
                    for label, reply in terminal_query_replies(wire, answered_queries):
                        os.write(master, reply)
                        append_audit(audit_path, "terminal-query-answered", label=label)
                    state = state_from_bytes(wire)
                    safe_state = _safe_state(state)
                    signature = {key: value for key, value in safe_state.items() if key != "bytesObserved"}
                    if signature != last_state:
                        append_audit(audit_path, "terminal-state-change", state=safe_state)
                        last_state = signature
                        last_change = time.monotonic()
                    if (authorization_code_submitted and not color_scheme_continued and
                            may_continue_color_scheme(state)):
                        os.write(master, b"\r")
                        color_scheme_continued = True
                        append_audit(audit_path, "native-color-scheme-continued", input="Enter", purpose="visual-appearance-only")
                        continue
                    if is_ready(state) or state.get("errorObserved"):
                        return state, enter_sent
                    if not enter_sent and may_press_enter(state):
                        append_audit(audit_path, "explicit-native-login-enter-confirmed", input="Enter", caseRoot=str(case_root))
                        os.write(master, b"\r")
                        enter_sent = True
                        append_audit(audit_path, "single-enter-sent", input="Enter")
            if authorization_code_fd is not None and authorization_code_fd in ready:
                # Re-check current eligibility at the actual read boundary.
                if (state is None or not enter_sent or
                        state.get("authorizationCodePromptObserved") is not True or
                        state.get("terminalAmbiguityObserved") or state.get("localPromptObserved")):
                    continue
                try:
                    code_chunk = os.read(
                        authorization_code_fd,
                        MAX_AUTHORIZATION_CODE_BYTES + 1 - len(code_buffer),
                    )
                except BlockingIOError:
                    continue
                if not code_chunk:
                    continue
                if _validate_and_submit_code(
                    master, code_buffer, code_chunk, audit_path, authorization_code_fd
                ):
                    authorization_code_submitted = True
            if (authorization_code_file is not None and can_read_code and state is not None and
                    state.get("authorizationCodePromptObserved") is True and
                    state.get("terminalAmbiguityObserved") is False and
                    state.get("localPromptObserved") is False):
                code_data, identity = read_authorization_code_file(authorization_code_file)
                if code_data is not None:
                    if _validate_and_submit_code(
                        master, code_buffer, code_data, audit_path,
                        before_submit=lambda: _unlink_code_before_submit(
                            authorization_code_file, identity
                        ),
                    ):
                        authorization_code_submitted = True
        return state_from_bytes(b"".join(chunks)), enter_sent
    finally:
        code_buffer[:] = b"\x00" * len(code_buffer)


def _record_dependencies(audit_root):
    entries = {}
    paths = {
        "prepare_agy_login.py": Path(__file__).resolve(),
        "probe_cursor_fresh_cli.py": Path(CURSOR.__file__).resolve(strict=True),
        "probe-agy-macos-interactive-manager.py": Path(AGY.__file__).resolve(strict=True),
        "native_startup_diagnostics.py": Path(AGY._diagnostics.__file__).resolve(strict=True),
        "native_terminal.py": Path(AGY._terminal.__file__).resolve(strict=True),
        "probe-agy-interactive-manager.py": Path(AGY.agy.__file__).resolve(strict=True),
        "probe-agy-linux-container.py": Path(AGY.agy.base.__file__).resolve(strict=True),
    }
    for name, path in paths.items():
        content = path.read_bytes()
        digest = sha(content)
        target = Path(audit_root) / name
        target.write_bytes(content)
        target.chmod(0o600)
        entries[name] = digest
    return entries


def _preflight(root, audit_path, sbpl):
    command = ["/usr/bin/sandbox-exec", "-p", sbpl, "/bin/sleep", "30"]
    proc = subprocess.Popen(command, cwd=root, env=env_for(root), stdin=subprocess.DEVNULL,
                            stdout=subprocess.DEVNULL, stderr=subprocess.DEVNULL,
                            start_new_session=True, close_fds=True)
    try:
        pgid = CURSOR.audit_spawned_child(root, proc, "agy-login-signal-preflight")
        append_audit(audit_path, "preflight-owned", pid=proc.pid, pgid=pgid)
        AGY.terminate_owned_group(proc, pgid, "agy-login-signal-preflight")
        append_audit(audit_path, "preflight-clean", pid=proc.pid, pgid=pgid,
                     returnCode=proc.returncode)
        return {"pid": proc.pid, "pgid": pgid, "terminated": proc.poll() is not None}
    except Exception:
        if proc.poll() is None:
            try:
                AGY.terminate_owned_group(proc, proc.pid, "agy-login-preflight-failure")
            except Exception:
                pass
        append_audit(audit_path, "preflight-failed", pid=proc.pid, parentReaped=proc.poll() is not None)
        raise


def run_login(root, audit_path, cli, timeout=MAX_LOGIN_SECONDS, *, authorization_code_fd=None,
              authorization_code_file=None):
    """Start the copied CLI behind a persisted process-identity gate."""
    if authorization_code_fd is not None and authorization_code_file is not None:
        raise RuntimeError("choose one authorization-code input method")
    if authorization_code_file is not None:
        authorization_code_file = validate_authorization_code_file(authorization_code_file)
    master, slave = pty.openpty()
    os.set_blocking(master, False)
    fcntl.ioctl(slave, termios.TIOCSWINSZ, struct.pack("HHHH", LOGIN_HEIGHT, LOGIN_WIDTH, 0, 0))
    tty_device = os.ttyname(slave)
    profile = sandbox_profile(root, cli, tty_device)
    gate = Path(root) / "native-start.gate"
    runner_started = Path(root) / "runner-started"
    exec_attempted = Path(root) / "native-exec-attempted"
    runner = (
        "import os,pathlib,sys,time\n"
        "gate=pathlib.Path(sys.argv[1]); started=pathlib.Path(sys.argv[2]); "
        "attempted=pathlib.Path(sys.argv[3]); exe=sys.argv[4]\n"
        "def mark(path, value):\n"
        "  with path.open('x', encoding='ascii') as f:\n"
        "    f.write(value+'\\n'); f.flush(); os.fsync(f.fileno())\n"
        "mark(started,'runner-started')\n"
        "while not gate.exists(): time.sleep(0.025)\n"
        "mark(attempted,'native-exec-attempted')\n"
        "os.execv(exe,[exe])"
    )
    interpreter = str(Path(sys.executable).resolve(strict=True))
    command = ["/usr/bin/sandbox-exec", "-p", profile, interpreter, "-B", "-c", runner,
               str(gate), str(runner_started), str(exec_attempted), str(cli)]
    proc = None
    pgid = None
    close_result = {"masterClosed": False, "slaveClosed": False}
    state = None
    enter_sent = False
    started = time.monotonic()
    stdin_attributes = None
    try:
        if authorization_code_fd is not None:
            stdin_attributes = _disable_terminal_echo(authorization_code_fd)
        proc = subprocess.Popen(command, cwd=root, env=env_for(root), stdin=slave, stdout=slave,
                                stderr=slave, start_new_session=True,
                                preexec_fn=AGY.claim_controlling_tty, close_fds=True)
        os.close(slave)
        close_result["slaveClosed"] = True
        pgid = CURSOR.audit_spawned_child(root, proc, "agy-private-login")
        append_audit(audit_path, "native-start-gated", pid=proc.pid, pgid=pgid,
                     command="antigravity", timeoutSeconds=timeout)
        with gate.open("xb") as stream:
            stream.write(b"go\n")
            stream.flush()
            os.fsync(stream.fileno())
        deadline = time.monotonic() + min(max(1, timeout), MAX_LOGIN_SECONDS)
        state, enter_sent = _capture_login(master, deadline, enter_sent=False,
                                           audit_path=audit_path, case_root=root,
                                           authorization_code_fd=authorization_code_fd,
                                           authorization_code_file=authorization_code_file)
        runner_seen = runner_started.is_file()
        exec_seen = exec_attempted.is_file()
        append_audit(audit_path, "native-login-observed", pid=proc.pid, pgid=pgid,
                     state=_safe_state(state), enterSent=enter_sent,
                     runnerStarted=runner_seen, nativeExecAttempted=exec_seen,
                     durationSeconds=round(time.monotonic() - started, 2))
        return {"promptReady": is_ready(state), "state": _safe_state(state), "enterSent": enter_sent,
                "runnerStarted": runner_seen, "nativeExecAttempted": exec_seen,
                "durationSeconds": round(time.monotonic() - started, 2),
                "exitCode": proc.poll()}
    finally:
        try:
            os.close(master)
            close_result["masterClosed"] = True
        except OSError:
            pass
        try:
            if proc is not None and pgid is not None:
                AGY.terminate_owned_group(proc, pgid, "agy-private-login")
                append_audit(audit_path, "native-login-cleanup", pid=proc.pid, pgid=pgid,
                             cleaned=proc.poll() is not None, closeResult=close_result)
            elif proc is not None:
                CURSOR.direct_child_cleanup(proc, grace=1.0)
                append_audit(audit_path, "native-login-cleanup-failed-audit", pid=proc.pid,
                             parentReaped=proc.poll() is not None, closeResult=close_result)
                raise RuntimeError("process identity audit failed; login result is invalid")
        finally:
            if not close_result["slaveClosed"]:
                try:
                    os.close(slave)
                    close_result["slaveClosed"] = True
                except OSError:
                    pass
            if stdin_attributes is not None:
                _restore_terminal(authorization_code_fd, stdin_attributes)


def _safe_state(state):
    keys = ("authBarrierObserved", "localPromptObserved", "readyHintObserved",
            "welcomeLabelObserved", "workspaceLabelObserved", "modelSelectionLabelObserved",
            "errorObserved", "errorCategories", "bytesObserved",
            "googleSignInOptionObserved", "nativeLoginChoiceObserved", "enterHintObserved",
            "authorizationCodePromptObserved", "codeFieldDiagnostics", "authScreenFixedRows", "terminalAmbiguityObserved", "terminalAmbiguitySite",
            "loginRowDiagnostics", "startupDiagnostics", "colorSchemeScreenObserved",
            "colorSchemeContinueObserved", "errorRowDiagnostics", "trustPromptObserved",
            "termsAcceptancePromptObserved", "codeFieldGeometry")
    return {key: state.get(key) for key in keys}


def _write_invalid_marker(root, marker):
    marker["valid"] = False
    _atomic_json(Path(root) / "owned-session.json", marker)


def invalidate_and_remove_private_home(root, marker):
    """Attempt marker invalidation and HOME removal even if either operation fails."""
    root = Path(root)
    home = root / "home"
    failures = []
    try:
        _write_invalid_marker(root, marker)
    except Exception as error:
        failures.append(type(error).__name__)
        try:
            (root / "owned-session.json").unlink(missing_ok=True)
        except Exception as unlink_error:
            failures.append(type(unlink_error).__name__)
    finally:
        try:
            if home.exists():
                shutil.rmtree(home)
        except Exception as error:
            failures.append(type(error).__name__)
    if home.exists():
        failures.append("HomeStillExists")
    if failures:
        raise RuntimeError("private HOME invalidation/removal failed")


def _prepare(binary=PINNED_BINARY, *, timeout=MAX_LOGIN_SECONDS, authorization_code_stdin=False,
             authorization_code_file=None):
    """Run one bounded native-login preparation; returns sanitized status only."""
    if sys.platform != "darwin" or os.uname().machine not in ("x86_64", "i386"):
        raise RuntimeError("AGY login preparation is pinned for Darwin x64")
    audit_root = _owned_dir("aok-agy-login-audit-")
    root = _owned_dir("aok-agy-login-")
    audit_path = audit_root / "audit.jsonl"
    global AUDIT_PATH
    AUDIT_PATH = audit_path
    AGY.AUDIT_PATH = audit_path
    CURSOR.AUDIT_PATH = audit_path
    created = datetime.now(timezone.utc)
    session_id = str(uuid.uuid4())
    marker = session_marker(session_id, created)
    _atomic_json(root / "owned-session.json", marker)
    dependencies = _record_dependencies(audit_root)
    source_hash = dependencies["prepare_agy_login.py"]
    append_audit(audit_path, "probe-prepared", sessionId=session_id,
                 sourceHashes=dependencies, sourceSHA256=source_hash,
                 binarySHA256=PINNED_BINARY_SHA256, terminalGeometry={"columns": LOGIN_WIDTH, "rows": LOGIN_HEIGHT}, authenticationMode="simulated-ssh" if MANUAL_OAUTH_MODE else "local", rootCreatedAt=marker["createdAt"],
                 rootExpiresAt=marker["expiresAt"])
    result = {"status": "blocked", "root": str(root), "auditRoot": str(audit_root),
              "sessionId": session_id,
              "backendAuthentication": "not-independently-verified"}
    print(json.dumps({"status": "pending", "root": str(root),
                      "auditRoot": str(audit_root)}, sort_keys=True), flush=True)
    try:
        binary = Path(binary).resolve(strict=True)
        if file_sha(binary) != PINNED_BINARY_SHA256:
            raise RuntimeError("pinned AGY binary hash mismatch")
        cli = root / "antigravity"
        shutil.copyfile(binary, cli)
        cli.chmod(0o500)
        if file_sha(cli) != PINNED_BINARY_SHA256:
            raise RuntimeError("copied AGY binary hash mismatch")
        profile = sandbox_profile(root, cli, "/dev/tty")
        preflight = _preflight(root, audit_path, profile)
        result["preflight"] = preflight
        observation = run_login(
            root, audit_path, cli, timeout=timeout,
            authorization_code_fd=0 if authorization_code_stdin else None,
            authorization_code_file=authorization_code_file,
        )
        result["observation"] = observation
        if not observation["promptReady"]:
            invalidate_and_remove_private_home(root, marker)
            try:
                cli.unlink()
            except OSError:
                pass
            append_audit(audit_path, "login-preparation-failed", sessionId=session_id,
                         reason="ready-prompt-not-observed", state=observation["state"])
            return result
        marker["valid"] = True
        marker["readyPromptObserved"] = True
        _atomic_json(root / "owned-session.json", marker)
        append_audit(audit_path, "login-preparation-prompt-ready", sessionId=session_id,
                     state=observation["state"], expiresAt=marker["expiresAt"])
        result["status"] = "prompt-ready"
        return result
    except BaseException as error:
        cleanup_failed = False
        try:
            invalidate_and_remove_private_home(root, marker)
        except Exception:
            cleanup_failed = True
        try:
            (root / "antigravity").unlink()
        except OSError:
            pass
        try:
            append_audit(audit_path, "login-preparation-failed", sessionId=session_id,
                         reasonType=type(error).__name__, cleanupFailed=cleanup_failed)
        except Exception:
            pass
        if not isinstance(error, Exception):
            raise
        result["reasonType"] = type(error).__name__
        if cleanup_failed:
            result["cleanupFailed"] = True
        return result


def prepare(binary=PINNED_BINARY, *, timeout=MAX_LOGIN_SECONDS, authorization_code_stdin=False,
            authorization_code_file=None):
    """Validate and always clean a private authorization-code file path."""
    if authorization_code_stdin and authorization_code_file is not None:
        raise RuntimeError("choose one authorization-code input method")
    validated_file = None
    try:
        if authorization_code_file is not None:
            candidate = Path(authorization_code_file).absolute()
            try:
                validated_file = validate_authorization_code_file(candidate)
            except RuntimeError:
                # A private, owned, single-link regular file may be rejected
                # only for unsafe mode/size. Remove that entry without ever
                # following a symlink or touching a path outside the task area.
                try:
                    entry = candidate.lstat()
                    if (stat.S_ISREG(entry.st_mode) and entry.st_uid == os.getuid() and
                            entry.st_nlink == 1):
                        remove_authorization_code_file(candidate)
                except (OSError, RuntimeError):
                    pass
                raise
        return _prepare(
            binary, timeout=timeout, authorization_code_stdin=authorization_code_stdin,
            authorization_code_file=validated_file,
        )
    finally:
        remove_authorization_code_file(validated_file)


def main(argv=None):
    parser = argparse.ArgumentParser(description=__doc__)
    parser.add_argument("--binary", type=Path, default=PINNED_BINARY)
    parser.add_argument("--timeout", type=int, default=MAX_LOGIN_SECONDS)
    code_input = parser.add_mutually_exclusive_group()
    code_input.add_argument("--authorization-code-stdin", action="store_true",
                            help="read one code from a TTY after the native code-entry prompt")
    code_input.add_argument("--authorization-code-file", type=Path,
                            help="read one private temporary code file only after the native prompt")
    parser.add_argument("--manual-oauth", action="store_true", help="exercise the documented SSH code flow using synthetic SSH environment metadata")
    args = parser.parse_args(argv)
    global MANUAL_OAUTH_MODE
    MANUAL_OAUTH_MODE = args.manual_oauth
    if args.timeout < 1 or args.timeout > MAX_LOGIN_SECONDS:
        parser.error(f"--timeout must be 1..{MAX_LOGIN_SECONDS} seconds")
    result = prepare(args.binary, timeout=args.timeout,
                     authorization_code_stdin=args.authorization_code_stdin,
                     authorization_code_file=args.authorization_code_file)
    print(json.dumps({"status": result["status"], "root": result["root"],
                      "auditRoot": result["auditRoot"]}, sort_keys=True))
    return 0 if result["status"] == "prompt-ready" else 1


if __name__ == "__main__":
    raise SystemExit(main())
