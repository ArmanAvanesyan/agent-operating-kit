"""One isolated native onboarding run; explicit file-enum input, no model turns."""
from pathlib import Path
import importlib.util,json,os,re,stat,sys,time,hashlib
spec=importlib.util.spec_from_file_location('agy_operator_helper',Path.cwd()/'scripts/native-loading/prepare_agy_login.py')
m=importlib.util.module_from_spec(spec);spec.loader.exec_module(m)
control=Path(sys.argv[1]); code=Path(sys.argv[2]); snapshot=None; master_fd=None; last_signature=None; last_change=0; actions=0
original_state=m.state_from_bytes; original_capture=m._capture_login; original_select=m.select.select

def public_screen(data):
 global snapshot,last_signature,last_change
 state=original_state(data)
 if state.get('terminalAmbiguityObserved'):
  snapshot=None;return state
 text=m.AGY.terminal_text(data,width=m.LOGIN_WIDTH,height=m.LOGIN_HEIGHT)
 signature=hashlib.sha256(text.encode()).hexdigest()
 if signature!=last_signature:
  last_signature=signature;last_change=time.monotonic()
  if not state.get('authBarrierObserved'):
   if 'Terms of Service & Data Use' in text:
    start=next(i for i,x in enumerate(text.splitlines()) if 'Terms of Service & Data Use' in x)
    public=text.splitlines()[start:start+22]
    safe=[]
    for row in public:
     row=re.sub(r'https?://\S+','[link]',row)
     row=re.sub(r'\b[^\s@]+@[^\s@]+\.[^\s@]+','[account]',row)
     row=re.sub(r'4/0[A-Za-z0-9_-]+|ya29\.[A-Za-z0-9_-]+','[redacted]',row)
     safe.append(row)
    print(json.dumps({'publicTermsScreen':safe,'screenSHA256':signature}),flush=True)
   elif not state.get('colorSchemeScreenObserved') and not m.is_ready(state):
    fixed={'choose','rendering','mode','altscreen','inline','trust','workspace','directory','folder','yes','no','allow','enter','continue','done','select','navigate','up','down','tab','space','security','warning','cancel'}
    print(json.dumps({'onboardingFixedRows':[{'row':i,'words':[x for x in re.findall(r'[A-Za-z]+',row.lower()) if x in fixed],'otherWords':sum(x not in fixed for x in re.findall(r'[A-Za-z]+',row.lower()))} for i,row in enumerate(text.splitlines()) if any(x in fixed for x in re.findall(r'[A-Za-z]+',row.lower()))]}),flush=True)
 snapshot=(state,text,signature)
 return state

def select_control(r,w,x,timeout):
 global actions,snapshot,last_change
 ready=original_select(r,w,x,timeout)
 if ready[0] or snapshot is None or time.monotonic()-last_change<0.6 or not control.exists():return ready
 info=control.lstat()
 if not stat.S_ISREG(info.st_mode) or info.st_uid!=os.getuid() or info.st_nlink!=1 or info.st_mode & 0o077 or info.st_size>512:raise RuntimeError('unsafe operator control')
 command=json.loads(control.read_text());control.unlink()
 state,text,signature=snapshot
 action=command.get('action');expected=command.get('screenSHA256')
 if expected!=signature or actions>=12 or state.get('authBarrierObserved') or state.get('terminalAmbiguityObserved') or state.get('localPromptObserved') or state.get('errorObserved'):raise RuntimeError('operator action blocked')
 terms=('Terms of Service & Data Use' in text and 'Google to collect and use my Interactions data,' in text and 'Done' in text and not state.get('colorSchemeScreenObserved'))
 if not terms or action not in {'terms-tab','terms-shift-tab'}:raise RuntimeError('unrecognized consent action')
 value={'terms-tab':b'\t','terms-shift-tab':b'\x1b[Z'}[action]
 m.append_audit(m.AUDIT_PATH,'operator-consent-navigation',action=action,screenSHA256=signature)
 os.write(master_fd,value);actions+=1;snapshot=None;last_change=time.monotonic()
 return ready

def capture(master,*args,**kwargs):
 global master_fd
 master_fd=master
 return original_capture(master,*args,**kwargs)
m.state_from_bytes=public_screen;m._capture_login=capture;m.select.select=select_control
# Only this approved fresh OAuth run uses manual code entry. No profile resume.
m.MANUAL_OAUTH_MODE=True
result=m.prepare(timeout=300,authorization_code_file=code,retain_onboarding=False)
print(json.dumps({'status':result['status'],'root':result['root'],'auditRoot':result['auditRoot'],'operatorActions':actions}),flush=True)
